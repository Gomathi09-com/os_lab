// LRU Page Replacement with Frame Table
#include <stdio.h>

int main() {

    int n,f,fault=0,hit=0,count=0;

    printf("Enter number of pages: ");
    scanf("%d",&n);

    int p[n];

    printf("Enter pages: ");
    for(int i=0;i<n;i++)
        scanf("%d",&p[i]);

    printf("Enter number of frames: ");
    scanf("%d",&f);

    int frame[f],time[f];

    for(int i=0;i<f;i++) {
        frame[i]=-1;
        time[i]=0;
    }

    printf("\nPage\tFrames\n");

    for(int i=0;i<n;i++) {

        int found=0;

        for(int j=0;j<f;j++) {

            if(frame[j]==p[i]) {

                hit++;
                count++;
                time[j]=count;
                found=1;
            }
        }

        if(!found) {

            int lru=0;

            for(int j=1;j<f;j++) {

                if(time[j]<time[lru])
                    lru=j;
            }

            count++;

            frame[lru]=p[i];
            time[lru]=count;

            fault++;
        }

        // Display frame table
        printf("%d\t",p[i]);

        for(int j=0;j<f;j++) {

            if(frame[j]==-1)
                printf("- ");
            else
                printf("%d ",frame[j]);
        }

        printf("\n");
    }

    printf("\nPage Faults = %d\n",fault);
    printf("Page Hits = %d\n",hit);

    return 0;
}
